Drop it into any agent

Last Look is a standard Solana signer that holds no key. Anything that accepts a signer, including x402 payments and your own transfers, is guarded with no other changes.

// The agent process never sees a private key.
import { createFirewallSigner } from "./client/firewallSigner.ts";

const signer = createFirewallSigner(agentAddress, () => ({
  tool: "fetch_paid_api",
  purpose: "premium rates data",
  tainted: agentHasReadUntrustedContent,
}));

// Use it anywhere a signer goes. A blocked payment throws, with the reasons.
await sendUsdc(signer, vendor, 0.05);
